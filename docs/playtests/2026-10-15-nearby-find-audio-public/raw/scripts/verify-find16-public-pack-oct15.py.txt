import datetime, json, os, pathlib, subprocess, uuid

root = pathlib.Path(r'D:\games\youjia-test\assistant-recovery')
out = root / 'find16-public-bb04-oct15'
verified = json.loads((out / 'verified.json').read_text(encoding='utf-8'))
assert verified['source'] == 'bb04fde115877380e3fda8fa80b96ee8753d7180'
pck = out / (verified['entry'] + '.pck')
profile = out / ('native-pack-profile-' + uuid.uuid4().hex)
profile.mkdir()
env = os.environ.copy()
env['APPDATA'] = str(profile)
env['LOCALAPPDATA'] = str(profile / 'local')
command = [r'D:\games\youjia-test\producer-recovery\godot472\Godot_v4.7.2-stable_win64_console.exe', '--headless', '--path', str(out), '--main-pack', str(pck), '--script', str(root / 'verify-find16-public-pack-oct15.gd')]
started = datetime.datetime.now(datetime.timezone.utc).isoformat()
result = subprocess.run(command, cwd=out, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=90)
(out / 'actual-public-packed-checks.log').write_bytes(result.stdout)
text = result.stdout.decode('utf-8-sig', errors='replace')
assert result.returncode == 0 and 'FIND16_STAMPED_PACK checks=64 failures=0 identity=bb04fde115877380e3fda8fa80b96ee8753d7180' in text, text
assert 'SCRIPT ERROR' not in text and 'ERROR:' not in text, text
record = {'source': verified['source'], 'pck': pck.name, 'command': command, 'isolatedProfile': str(profile), 'startedAt': started, 'exit': result.returncode, 'checks': 64, 'status': 'PASS', 'method': 'Read the actual downloaded public PCK with the standalone SceneTree helper; inspect its identity/announcement/glyphs/approved PCM bytes and nonloop format. AppData scoped only to this child process.'}
(out / 'actual-public-packed-checks.json').write_text(json.dumps(record, indent=2) + '\n', encoding='utf-8')
print(text, flush=True)
