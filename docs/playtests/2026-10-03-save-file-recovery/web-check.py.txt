import json
from playwright.sync_api import sync_playwright
with sync_playwright() as p:
 b=p.chromium.launch(headless=True,executable_path='/usr/bin/chromium',args=['--no-sandbox','--use-gl=angle','--use-angle=swiftshader','--enable-webgl'])
 page=b.new_page(viewport={'width':1280,'height':720});errors=[]
 page.on('pageerror',lambda e:errors.append(str(e)))
 page.on('console',lambda m:errors.append(m.text) if m.type=='error' else None)
 url='http://127.0.0.1:8789/controlled/index.html'
 results=[]
 for stage in ['staged','recovered','resaved']:
  if not results:page.goto(url,wait_until='domcontentloaded',timeout=30000)
  else:page.reload(wait_until='domcontentloaded',timeout=30000)
  page.wait_for_function('window.__saveRecovery !== undefined',timeout=120000)
  r=page.evaluate('window.__saveRecovery');results.append(r)
  assert r['phase']==stage and not r['failures'],r
 page.wait_for_timeout(1000)
 page.screenshot(path='/tmp/youjia-save-web-recovered.png')
 assert not errors,errors
 print(json.dumps({'phases':results,'errors':errors}),flush=True)
 open('/tmp/youjia-save-web-result.json','w').write(json.dumps({'phases':results,'errors':errors}))
 b.close()
