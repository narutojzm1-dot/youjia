extends SceneTree
# Throwaway render probe (not committed): capture the bottom notice over sun/overcast.
var main
func _initialize(): call_deferred("run")
func run():
	var out := OS.get_environment("YJ_OUT")
	for dims in [Vector2i(1280,720), Vector2i(390,844)]:
		root.size = dims
		DisplayServer.window_set_size(dims)
		main = load("res://scenes/main.tscn").instantiate()
		root.add_child(main)
		await process_frame; await process_frame
		main._start_holiday()
		for weather in ["sun","overcast"]:
			main._world.set_weather(weather)
			for i in 30: await process_frame
			main._show_notice_key("notice.first_hint")
			main._notice.add_theme_font_size_override("font_size", 18)
			main._fit_notice()
			for i in 6: await process_frame
			var img := root.get_texture().get_image()
			img.save_png("%s/long-%dx%d-%s.png" % [out, dims.x, dims.y, weather])
			print("saved ", dims, weather, " notice_rect=", Rect2(main._notice.get_global_transform_with_canvas().origin, main._notice.size))
		main.queue_free()
		await process_frame
	quit(0)
