extends Node
var main
var world
var stage := -1
func _ready():
 main = load("res://scenes/main.tscn").instantiate()
 add_child(main)
 await get_tree().process_frame
 main._on_play_pressed()
 await get_tree().process_frame
 main.set_process(false)
 world = main._world
 world.set_process(false)
 JavaScriptBridge.eval("window.petStage=0;window.petReady=true")
func _process(_delta):
 if world == null: return
 var next = int(JavaScriptBridge.eval("window.petStage"))
 if next == stage: return
 stage = next
 var ids = ["cow", "horse", "sheep_a", "sheep_b"]
 var id = ids[stage % 4]
 var animal = world.actor_named(id)
 TuningStore.set_value("ui.reduced_motion", stage >= 4)
 animal._pet_ack_cooldown = 0.0
 world.debug_place_player(animal.position + Vector2(30, 0))
 world._interact_with_target("pet:" + id)
 main._refresh_hud()
 var result = {"stage":stage,"actor":id,"pose":animal._posture_id,"remaining":animal._ack_left,"heart":world._effects_overlay.pet_feedback_snapshot(),"reduced":stage>=4}
 JavaScriptBridge.eval("window.petResult=" + JSON.stringify(result))
