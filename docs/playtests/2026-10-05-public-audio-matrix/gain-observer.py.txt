import json,time,sys
from pathlib import Path
from playwright.sync_api import sync_playwright
out=Path(sys.argv[2]);out.mkdir(parents=True,exist_ok=True)
errors=[];events=[]
with sync_playwright() as p:
 b=p.chromium.launch(executable_path='/usr/bin/chromium',headless=True,args=['--no-sandbox','--use-gl=angle','--use-angle=swiftshader','--enable-webgl'])
 pg=b.new_page(viewport={'width':1280,'height':720})
 pg.on('pageerror',lambda e:errors.append(str(e)));pg.on('console',lambda m:errors.append(m.text) if m.type=='error' else None)
 pg.add_init_script("""window.first=false;window.addEventListener('youjia:first-frame',()=>window.first=true);
 window.__observedGainNodes=[];const proto=window.BaseAudioContext?.prototype;
 if(proto){const original=proto.createGain;proto.createGain=function(...args){const node=Reflect.apply(original,this,args);window.__observedGainNodes.push({node,creator:new Error().stack});return node;};}
 """)
 pg.goto(sys.argv[1]);pg.wait_for_function('window.first',timeout=120000);build=pg.locator('html').get_attribute('data-build')
 pg.mouse.click(640,368);pg.wait_for_timeout(2000);pg.mouse.click(1165,44);pg.wait_for_timeout(2000)
 def snap(name):
  d=pg.evaluate('({backend:JSON.parse(window.__manusBgm.diagnostics()),gains:window.__observedGainNodes.map((o,i)=>({id:i,gain:o.node.gain.value,creator:o.creator}))})');events.append({'stage':name,**d});pg.screenshot(path=str(out/(name+'.png')));return {x['id']:x['gain'] for x in d['gains']}
 initial=snap('initial-paused')
 pg.mouse.click(477,419);pg.wait_for_timeout(1000);music_zero=snap('music-zero')
 pg.mouse.click(640,419);pg.wait_for_timeout(1000);music_half=snap('music-half')
 pg.mouse.click(477,554);pg.wait_for_timeout(1000);env_zero=snap('ambience-zero')
 pg.mouse.click(640,554);pg.wait_for_timeout(1000);env_half=snap('ambience-half')
 pg.mouse.click(640,604);pg.wait_for_timeout(1000);muted=snap('master-muted')
 pg.mouse.click(640,604);pg.wait_for_timeout(1000);restored=snap('master-restored')
 def zero_change(before,after):return [i for i in before if before[i]>0.00001 and after.get(i)==0.0]
 backend_ids=[x['id'] for x in events[0]['gains'] if 'channel' in x['creator']]
 music=[i for i in zero_change(initial,music_zero) if i in backend_ids];ambience=[i for i in zero_change(music_half,env_zero) if i in backend_ids]
 (out/'raw-events.json').write_text(json.dumps({'build':build,'events':events,'errors':errors},ensure_ascii=False,indent=2)+'\n')
 assert len(music)==1 and len(ambience)==1 and music!=ambience,(music,ambience)
 mid=music[0];eid=ambience[0]
 assert music_zero[eid]==initial[eid] and env_zero[mid]==music_half[mid]
 assert 0<music_half[mid]<initial[mid] and 0<env_half[eid]<music_half[eid]
 assert muted[mid]==muted[eid]==0 and restored[mid]>0 and restored[eid]>0
 assert restored[mid]==env_half[mid] and restored[eid]==env_half[eid]
 assert not errors,errors
 (out/'result.json').write_text(json.dumps({'build':build,'url':sys.argv[1],'utc':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),'events':events,'music_node_id':mid,'ambience_node_id':eid,'errors':errors,'checks_passed':True,'scope':'public real mouse actions; observation-only createGain wrapper returns original node without changing graph/control/PCM; reads actual AudioParam gain.value, not fake players or ear/hearing proof'},ensure_ascii=False,indent=2)+'\n');b.close()
