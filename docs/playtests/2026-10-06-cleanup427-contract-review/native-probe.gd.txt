extends SceneTree
const Native = preload("res://scripts/persistence/native_save_host.gd")
const Codec = preload("res://scripts/persistence/save_data_codec.gd")
class TestStore extends "res://autoload/save_store.gd":
 func _ready(): pass
func _initialize(): call_deferred("run")
func run():
 var s := ExplorationSession.new(ExplorationRoutes.catalog(), 0)
 s.begin(ExplorationRoutes.NEAR_PATH, {"day":1,"elapsed":0.0}, 7)
 s.request_return("player")
 var expected: Dictionary = s.to_record()
 expected.session.started_clock.future_payload={"keep":"raw"}
 var snapshot := Codec.defaults()
 snapshot.exploration=expected.duplicate(true)
 snapshot.exploration_committed_serial=1
 var path := "user://pr427-native.json"
 var file := FileAccess.open(path, FileAccess.WRITE)
 file.store_string(JSON.stringify(snapshot));file.close()
 var backend := Native.new(path,"user://pr427-native.tmp","user://pr427-native.bak")
 var store := TestStore.new()
 root.add_child(store)
 store._backend=backend
 store._boot_state="ready"
 store._data=backend.get_initial_snapshot()
 print("COORDINATOR ",store._connect_coordinator(store._data,backend.get_initial_token(),Native.NAMESPACE))
 var original := FileAccess.get_file_as_bytes(path)
 var events=[]
 store.commit_rejected.connect(func(op,kind,code):events.append([op,kind,code]))
 store.commit_confirmed.connect(func(op,kind):events.append([op,kind,"confirmed"]))
 var host:=ExplorationHost.new(store)
 host.restore()
 print("RESTORE ",host.restore_action," RAW_BEFORE ",JSON.stringify(store.get_exploration_record()))
 await store.flush_pending()
 print("BACKEND_CALLS ",backend._calls," FILE_IDENTICAL ",FileAccess.get_file_as_bytes(path)==original)
 print("EVENTS ",JSON.stringify(events))
 print("RAW_AFTER ",JSON.stringify(store.get_exploration_record()))
 print("PRESERVED ",store.get_exploration_record()==expected)
 store.free()
 quit()
